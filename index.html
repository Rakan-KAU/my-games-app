// ذاكرة سحابية في السيرفر global.gamesDatabase = global.gamesDatabase || [ { id: 1, title: "Rocket League", platform: "PS5 / PC", price: 0 }, { id: 2, title: "FC 26", platform: "PS5", price: 69.99 } ]; module.exports = async function (context, req) { // إعدادات الـ CORS الكاملة لمنع أي حظر const headers = { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; // التعامل مع طلبات التصفح السريعة if (req.method === "OPTIONS") { context.res = { status: 200, headers: headers }; return; } // 1. جلب البيانات (GET) if (req.method === "GET") { context.res = { status: 200, headers: headers, body: global.gamesDatabase }; return; } // 2. الإضافة والحذف (POST) if (req.method === "POST") { const body = req.body || {}; // إذا كان الطلب حذف if (body.action === "delete") { global.gamesDatabase = global.gamesDatabase.filter(g => g.id != body.id); context.res = { status: 200, headers: headers, body: global.gamesDatabase }; return; } // إذا كان الطلب إضافة لعبة جديد if (body.title) { const newGame = { id: Date.now(), title: body.title, platform: body.platform, price: parseFloat(body.price) || 0 }; global.gamesDatabase.push(newGame); context.res = { status: 200, headers: headers, body: global.gamesDatabase }; return; } } };
